// ================= Basic shapes =================
 
// Box draw kore
void box(float x1,float y1,float x2,float y2,float x3,float y3,float x4,float y4,float xf,float yf)
{
    glBegin(GL_POLYGON);
    glVertex2f(x1+xf,y1+yf);
    glVertex2f(x2+xf,y2+yf);
    glVertex2f(x3+xf,y3+yf);
    glVertex2f(x4+xf,y4+yf);
    glEnd();
}
 
// Triangle draw kore
void tri(float x1,float y1,float x2,float y2,float x3,float y3,float xf,float yf)
{
    glBegin(GL_POLYGON);
    glVertex2f(x1+xf,y1+yf);
    glVertex2f(x2+xf,y2+yf);
    glVertex2f(x3+xf,y3+yf);
    glEnd();
}
 
// Circle draw kore
void circle(float radius,float xc,float yc,float r,float g,float b)
{
    glBegin(GL_POLYGON);
 
    for(int i=0;i<200;i++)
    {
        glColor3ub(r,g,b);
 
        float pi=3.1416;
        float A=(i*2*pi)/200;
        float x=radius*cos(A);
        float y=radius*sin(A);
 
        glVertex2f(x+xc,y+yc);
    }
 
    glEnd();
}
 
 
// Straight line
void line(float x1,float y1,float x2,float y2)
{
    glBegin(GL_LINES);
    glVertex2f(x1,y1);
    glVertex2f(x2,y2);
    glEnd();
}

// ================= Smoke =================
 
void smoke1_1()
{
    if(!showSmoke)
        return;
 
    for(int i=0;i<3;i++)
    {
        float phase=i*1.3f;
        float cycle=fmodf(smokeTime*0.6f+phase,3.0f);
        float px=-7.5f+0.15f*sinf(cycle*2.0f+phase);
        float py=1.6f+cycle;
        float size=0.22f-cycle*0.05f;
 
        if(size<0.05f)
            size=0.05f;
 
        float fadeT=cycle/3.0f;
        float cr=175+fadeT*(224-175);
        float cg=175+fadeT*(224-175);
        float cb=175+fadeT*(224-175);
 
        glColor3ub(cr,cg,cb);
        box(px-size,py-size,px+size,py-size,px+size,py+size,px-size,py+size,0,0);
    }
}

// ================= Reset =================
 
void resetScene1_1()
{
    // Sun
    sunOffsetY = -5.5f;
    sunUserControl = 0;
 
    // Sky
    skyR = 90;
    skyG = 105;
    skyB = 135;
 
    // Clouds
    cloud1OffX = 0.0f;
    cloud2OffX = 0.0f;
    showClouds = 1;
 
    // Flag
    showFlag = 1;
    flagTime = 0.0f;
    flagAngle = 0.0f;
 
    // Boat
    showBoat = 1;
    boatTime = 0.0f;
    boatBobY = 0.0f;
 
    // Smoke
    showSmoke = 1;
    smokeTime = 0.0f;
 
    // Tree
    leafTime = 0.0f;
 
    // River
    riverLineTime = 0.0f;
    riverLineBoost = 0.0f;
 
    // Animation
    animationSpeed = 0.015f;
 
    currentScene = 1;
}
 
 
// ================= Animation =================
 
void updateAnimation1_1(int value)
{
    // Sun rising
    if(!sunUserControl)
    {
        if(sunOffsetY < 0.0f)
        {
            sunOffsetY += animationSpeed;
 
            if(sunOffsetY > 0.0f)
                sunOffsetY = 0.0f;
        }
    }
 
    // ================= Sky color =================
 
    float sunY = 7.0f + sunOffsetY;
    float t = (sunY-1.5f) / (7.0f-1.5f);
 
    if(t < 0.0f)
        t = 0.0f;
    if(t > 1.0f)
        t = 1.0f;
 
    skyR = 90 + t*(158-90);
    skyG = 105 + t*(199-105);
    skyB = 135 + t*(224-135);
 
 
    // ================= Clouds =================
 
    if(showClouds)
    {
        cloud1OffX += 0.025f;
        cloud2OffX += 0.02f;
 
        if(cloud1OffX > 40.0f)
            cloud1OffX = -40.0f;
 
        if(cloud2OffX > 40.0f)
            cloud2OffX = -40.0f;
    }
 
 
    // ================= Flag =================
 
    if(showFlag)
    {
        flagTime += 0.03f;
        flagAngle = 12.0f*sinf(flagTime*3.0f);
    }
 
 
    // ================= Boat =================
 
    if(showBoat)
    {
        boatTime += 0.03f;
        boatBobY = 0.05f*sinf(boatTime*2.5f);
    }
 
 
    // ================= Smoke =================
 
    if(showSmoke)
        smokeTime += 0.03f;
 
 
    // ================= Leaves =================
 
    leafTime += 0.03f;
 
 
    // ================= River =================
 
    riverLineTime += 0.03f;
 
    if(riverLineBoost > 0.0f)
        riverLineBoost -= 0.01f;
 
    if(riverLineBoost < 0.0f)
        riverLineBoost = 0.0f;
 
 
    glutPostRedisplay();
    glutTimerFunc(30,updateAnimation1_1,0);
}
 
 
// ================= Mouse =================
 
void mouseClick1_1(int button,int state,int x,int y)
{
    if(state != GLUT_DOWN)
        return;
 
    // Screen theke world coordinate
    float wx = -20.0f + (x/1200.0f)*40.0f;
    float wy = 10.0f - (y/600.0f)*20.0f;
 
    // Left click
    if(button == GLUT_LEFT_BUTTON)
    {
        // Sun control
        if(wy > 1.5f)
        {
            sunUserControl = 1;
            sunOffsetY = wy - 7.0f;
 
            if(sunOffsetY > 0.0f)
                sunOffsetY = 0.0f;
 
            if(sunOffsetY < -5.5f)
                sunOffsetY = -5.5f;
        }
 
        // River click
        if(wy >= -0.5f && wy <= 1.5f)
        {
            riverLineBoost = 0.15f;
        }
    }
 
    // Right click
    else if(button == GLUT_RIGHT_BUTTON)
    {
        resetScene1_1();
    }
 
    glutPostRedisplay();
}
 
 
// ================= Keyboard =================
 
void keyboard1_1(unsigned char key,int x,int y)
{
    switch(key)
    {
        // Stop animation
        case 'a':
        case 'A':
            animationSpeed = 0.0f;
            break;
 
        // Start animation
        case 'w':
        case 'W':
            animationSpeed = 0.015f;
            break;
 
        // Cloud on/off
        case 'c':
        case 'C':
            showClouds = !showClouds;
            break;
 
        // Flag on/off
        case 'f':
        case 'F':
            showFlag = !showFlag;
            break;
 
        // Boat on/off
        case 'b':
        case 'B':
            showBoat = !showBoat;
            break;
 
        // Smoke on/off
        case 's':
        case 'S':
            showSmoke = !showSmoke;
            break;
 
        // Reset
        case 'r':
        case 'R':
            resetScene1_1();
            break;
 
        // Scene 2
        case 'n':
        case 'N':
            currentScene = 2;
            break;
 
        default:
            break;
    }
 
    glutPostRedisplay();
}
 
 
// ================= Special keys =================
 
void specialKeys1_1(int key,int x,int y)
{
    // Sun up
    if(key == GLUT_KEY_UP)
    {
        sunUserControl = 1;
        sunOffsetY += 0.3f;
 
        if(sunOffsetY > 0.0f)
            sunOffsetY = 0.0f;
    }
 
    // Sun down
    if(key == GLUT_KEY_DOWN)
    {
        sunUserControl = 1;
        sunOffsetY -= 0.3f;
 
        if(sunOffsetY < -5.5f)
            sunOffsetY = -5.5f;
    }
 
    glutPostRedisplay();
}
 
 
// ================= Main =================
 
int main(int argc,char** argv)
{
    glutInit(&argc,argv);
    glutInitDisplayMode(GLUT_SINGLE | GLUT_RGB);
    glutInitWindowSize(1200,600);
    glutInitWindowPosition(200,100);
 
    glutCreateWindow("Scene-1 : Gram");
 
    glutDisplayFunc(display1_1);
    glutMouseFunc(mouseClick1_1);
    glutKeyboardFunc(keyboard1_1);
    glutSpecialFunc(specialKeys1_1);
 
    glutTimerFunc(30,updateAnimation1_1,0);
 
    glClearColor(1,1,1,1);
 
    // Coordinate range
    gluOrtho2D(-20,20,-10,10);
 
    glutMainLoop();
 
    return 0;
}
